cmake_minimum_required(VERSION 3.16)
project(llama_logits_poc LANGUAGES C CXX)

set(LLAMA_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../.." CACHE PATH "Pinned llama.cpp source")
set(POC_GGML_LIB_DIR "" CACHE PATH "Optional directory containing the pinned shared ggml SDK libraries")
if(POC_GGML_LIB_DIR)
    get_filename_component(POC_GGML_LIB_DIR "${POC_GGML_LIB_DIR}" ABSOLUTE)
    get_filename_component(POC_GGML_INCLUDE_DIR "${LLAMA_SOURCE_DIR}/ggml/include" ABSOLUTE)
    foreach(component ggml-base ggml)
        set(component_library "${POC_GGML_LIB_DIR}/${CMAKE_SHARED_LIBRARY_PREFIX}${component}${CMAKE_SHARED_LIBRARY_SUFFIX}")
        if(NOT EXISTS "${component_library}")
            message(FATAL_ERROR "Pinned ggml library is missing: ${component_library}")
        endif()
        add_library(${component} SHARED IMPORTED GLOBAL)
        set_target_properties(${component} PROPERTIES
            IMPORTED_LOCATION "${component_library}"
            INTERFACE_INCLUDE_DIRECTORIES "${POC_GGML_INCLUDE_DIR}"
            INTERFACE_COMPILE_DEFINITIONS "GGML_SHARED;GGML_BACKEND_SHARED")
    endforeach()
    set_property(TARGET ggml PROPERTY INTERFACE_LINK_LIBRARIES ggml-base)
    # The upstream parent-target check skips its entire ggml/CUDA build.
    set(LLAMA_USE_SYSTEM_GGML OFF CACHE BOOL "" FORCE)
    message(STATUS "Using pinned prebuilt ggml libraries: ${POC_GGML_LIB_DIR}")
endif()
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_UI OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_HTML OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE_DIR}" llama-build)
find_package(Threads REQUIRED)
add_executable(llama-logits-poc main.cpp)
target_compile_features(llama-logits-poc PRIVATE cxx_std_17)
target_link_libraries(llama-logits-poc PRIVATE llama Threads::Threads)
